# Prima della pubblicazione occorre:
#   1. confermare gli slug reali con:  analizza_log.py --scopri
#   2. verificare che non blocchi URL utili con:
#        robots_check.py --robots attuale.txt --confronta robots_proposto.txt \
#                        --urls url_da_log.txt
#      Il comando esce con codice 2 se qualche URL utile risulta bloccata.
#   3. verificare se YITH ha le "pretty URL" attive (vedi NOTE_REGOLE.md,
#      punto 12): in quel caso i filtri NON sono query string e nessuna
#      regola qui sotto li intercetta.
#
# Principio guida: bloccare le URL inutili, mai le risorse di rendering.
# Motivazione regola per regola in NOTE_REGOLE.md

User-agent: *

# --- Area amministrativa e endpoint di WordPress -------------------------
# admin-ajax.php resta consentito: molti temi e YITH lo usano per caricare
# griglie prodotto e filtri via AJAX. Bloccarlo puo' impedire a Google di
# renderizzare correttamente le pagine categoria.
Disallow: /wp-admin/
Allow: /wp-admin/admin-ajax.php
Disallow: /wp-login.php
Disallow: /wp-register.php
Disallow: /xmlrpc.php
Disallow: /wp-cron.php
Disallow: /wp-trackback.php

# --- REST API -----------------------------------------------------------
# Due forme della stessa cosa: permalink e query string.
Disallow: /wp-json/
Disallow: /*?rest_route=

# --- wp-includes --------------------------------------------------------
# Gli Allow sono PIU' LUNGHI del Disallow, quindi vincono (regola del match
# piu' lungo). Servono: jQuery sta in /wp-includes/js/, e gli stili dei
# blocchi Gutenberg stanno in /wp-includes/blocks/ e /wp-includes/css/.
# Senza questi Allow si rischia di rompere il rendering.
Disallow: /wp-includes/
Allow: /wp-includes/js/
Allow: /wp-includes/css/
Allow: /wp-includes/blocks/

# --- Feed, trackback, commenti ------------------------------------------
# Serve SIA la forma alla radice SIA quella annidata: senza "/*/feed/"
# restano scansionabili /categoria-prodotto/x/feed/ e /prodotto/y/feed/.
Disallow: /feed/
Disallow: /*/feed/
Disallow: /comments/feed/
Disallow: /*/trackback/
Disallow: /trackback/
Disallow: /*?replytocom=

# --- Archivi autore -----------------------------------------------------
# VERIFICARE lo slug reale: su installazioni italiane puo' essere /autore/.
# Se l'archivio autore non esiste, queste righe sono inerti ma innocue.
Disallow: /author/
Disallow: /autore/

# --- Ricerca interna ----------------------------------------------------
# "/*?s=" copre la ricerca su qualunque percorso, non solo in home.
# "/*&s=" copre il parametro in mezzo alla query string.
Disallow: /*?s=
Disallow: /*&s=
Disallow: /search/
Disallow: /ricerca/

# --- Carrello, checkout, account ----------------------------------------
# VERIFICARE quali di questi slug esistono davvero.

Disallow: /cart/

# --- Azioni WooCommerce sul carrello ------------------------------------
# La forma "/*nome=" intercetta sia "?nome=" sia "&nome=" in qualunque
# posizione. Le forme "/?nome=" (solo home) e "/&nome=" (impossibile)
# della bozza precedente non bloccavano nulla sulle pagine reali.
Disallow: /*add-to-cart=
Disallow: /*removed_item=
Disallow: /*remove_item=
Disallow: /*undo_item=
Disallow: /*?wc-ajax=
Disallow: /*?wc-api=

# --- Ordinamento e densita' di risultati --------------------------------
Disallow: /*?orderby=
Disallow: /*&orderby=
Disallow: /*products-per-page=
Disallow: /*?per_page=
Disallow: /*&per_page=

Disallow: /*?stock_status=
